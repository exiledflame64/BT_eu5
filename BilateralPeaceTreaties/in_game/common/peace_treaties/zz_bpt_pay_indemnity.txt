# Bilateral Peace Treaties - Treaty group 2: the winner pays gold to the loser
# inside the same demand deal. Negative cost -> reduces the deal's net war score.
#
# KEPT IN ITS OWN FILE on purpose: spec constraint 5 (a deal must never contain
# both a gold demand and a gold offer) has NO engine-level enforcement hook
# (no exclusivity field in the treaty schema, no trigger exposing the deal under
# construction - script_docs 2026-07-19). To drop give-gold entirely, delete
# this file plus bpt_indemnity_loan_value - nothing else references them.
#
# WHY FIXED TIERS AND NOT A +/- STEPPER (Session A.7, data_types dump):
# the vanilla gold stepper is PeaceOfferWarScoreTreatyGlue, whose complete
# function set is Get*/Is*/On(Inc|Dec)*Click - there is no value setter, the
# 0-floor lives in IsDecWarScoreEnabled (C++), and the only glue instances the
# GUI can reach are GetGoldTreaty / GetSacrificeTreaty. A second stepper could
# therefore only drive the SAME gold treaty with the same floor. Scripted
# treaties are also not war-score treaties (IsWarScoreTreaty = no), so they
# cannot be rendered as a stepper at all.
# The earlier value-selector version rendered as one row per step (six
# near-identical "Pay indemnity" entries), so v1 ships three clearly named
# tiers instead - the spec's own fallback (spec 5.3).
#
# VALUATION mirrors vanilla exactly (measured Session A.2/A.4): the vanilla gold
# treaty is denominated in the payer's economy - 1 Loan = 5% war score = five
# months of the payer's trade+tax income, capped at 5 Loans / 25%. Each tier is
# therefore a whole number of loans, and its war-score cost is a constant.
# No treasury requirement: author decision 2026-07-19 - mirror the vanilla loan
# rule and let the payer go into debt.

bpt_pay_indemnity_small = {
	category = country

	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = 5
			multiply = -1
		}
	}

	potential = {
		bpt_human_involved_trigger = yes
		scope:winner = {
			monthly_income_trade_and_tax > 0
		}
	}

	allow = {
	}

	effect = {
		scope:winner = {
			add_gold = {
				value = bpt_indemnity_loan_value
				multiply = -1
			}
		}
		scope:loser = {
			add_gold = bpt_indemnity_loan_value
		}
	}

	ai_desire = {
		value = 0
	}
}
