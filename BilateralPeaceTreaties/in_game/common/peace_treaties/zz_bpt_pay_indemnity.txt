# Bilateral Peace Treaties - gold the winner pays to the loser inside the same
# demand deal. Negative cost -> reduces the deal's net war score.
#
# Five Identical one-loan units. Click one coin per Loan, up to five.
# 1 loan = 5% warscore.

bpt_pay_indemnity_1 = {
	category = country
	cost = { add = { desc = "DIPLOREASON_BASE" value = 5 multiply = -1 } }
	potential = {
		bpt_human_involved_trigger = yes
		scope:winner = { monthly_income_trade_and_tax > 0 }
	}
	allow = { }
	effect = {
		scope:winner = { add_gold = { value = bpt_indemnity_loan_value multiply = -1 } }
		scope:loser = { add_gold = bpt_indemnity_loan_value }
	}
	ai_desire = { value = 0 }
}
