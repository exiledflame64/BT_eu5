# Bilateral Peace Treaties - gold the winner pays to the loser inside the same
# demand deal. Negative cost -> reduces the deal's net war score.
#
# DESIGN: FIVE IDENTICAL ONE-LOAN UNITS, NOT ALTERNATIVE TIERS.
# The engine has no mutual-exclusion mechanism for scripted treaties
# (are_targets_exclusive only makes location/province targets exclusive with
# cede treaties - peace_treaties/readme.txt), so a 1/3/5-loan tier set could be
# selected all at once and add up to nonsense. Five interchangeable one-loan
# units have no invalid combination by construction: select N of them and the
# payment is N loans, 1..5 - exactly the vanilla range ("Maximum is 5 Loans",
# 5% war score each, 25% total). Each click adds a loan, which is also how the
# vanilla stepper behaves.
#
# All five carry the SAME cost (-5) on purpose: the give pane identifies gold
# entries by that exact value (EqualTo_CFixedPoint), since the engine exposes no
# treaty key, tag or category accessor to the GUI.
#
# VALUATION mirrors vanilla exactly (measured Session A.2/A.4): 1 Loan = 5% war
# score = five months of the payer's trade+tax income (bpt_indemnity_loan_value).
# No treasury requirement: author decision - mirror the vanilla loan rule and
# let the payer go into debt.
#
# WHY NOT A +/- STEPPER: the vanilla stepper drives PeaceOfferLateralView's one
# and only gold object, whose direction is fixed by the deal mode, and its glue
# type has no setter (DISCOVERY 20/21). To drop give-gold entirely, delete this
# file plus bpt_indemnity_loan_value - nothing else references them.

bpt_pay_indemnity_1 = {
	category = country
	cost = { add = { desc = "DIPLOREASON_BASE" value = 5 multiply = -1 } }
	potential = {
		bpt_human_involved_trigger = yes
		scope:winner = { monthly_income_trade_and_tax > 0 }
	}
	allow = { }
	effect = {
		scope:winner = { add_gold = { value = bpt_indemnity_loan_value multiply = -1 } }
		scope:loser = { add_gold = bpt_indemnity_loan_value }
	}
	ai_desire = { value = 0 }
}
