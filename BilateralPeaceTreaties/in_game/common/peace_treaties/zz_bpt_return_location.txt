# Bilateral Peace Treaties - Treaty 1: the winner cedes one of its own locations
# back to the loser inside the same demand deal. Negative cost -> reduces the
# deal's net war score. One entry per matching winner-owned location.
#
# Pattern source: disband_kontor / dismantle_fortifications (cost formula) and
# rtr_rein_in_rebellion (change_location_owner inside a treaty effect).

bpt_return_location = {
	category = location

	# Vanilla valuation, sign-flipped, ceding/receiving roles swapped:
	# vanilla demand pricing is scope:target.location_peace_cost(scope:loser|scope:winner)
	# on a LOSER-owned target; here the target is WINNER-owned and the loser receives.
	# Argument order (giver|taker) confirmed by script_docs (triggers.log).
	#
	# Session A.2 finding: both vanilla flows additionally multiply a location's
	# value by the TAKER's side conquer_cost factor of the war's wargoal (Adjud:
	# 1.78 base x 1.5 "Show Superiority" = 2.67 in the demand list AND the Offer
	# Tribute tab). Mirrored via bpt_taker_conquer_cost_factor
	# (script_values/bpt_values.txt); the taker of a returned location is scope:loser.
	# The factor table is per war-goal TYPE - the trigger accepts nothing finer
	# (Session A.4); see bpt_values.txt for exactness notes. The vanilla
	# component breakdown (tax base/development/...) is computed inside the
	# hardcoded valuation and is not exposed to script - the tooltip shows the
	# final number only.
	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = "scope:target.location_peace_cost(scope:winner|scope:loser)"
			multiply = bpt_taker_conquer_cost_factor
			multiply = -1
		}
	}

	# NO base_antagonism. It produced the right number and filled the antagonism
	# column natively, but Session A.19 settled where the engine sends it: to the
	# side COMPOSING the deal - i.e. it made the player hated for giving land
	# away. The engine has no way to know a scripted treaty benefits the loser.
	# The receiver's antagonism is therefore dropped explicitly in the effect,
	# using the same figure vanilla charges. Consequence: the antagonism column
	# shows 0.00 on give rows, which is the honest value for the giver - the
	# receiver's figure appears in the row's tooltip.

	potential = {
		bpt_human_involved_trigger = yes
	}

	allow = {
	}

	# Giver-agnostic: one entry per location instead of one per enemy war
	# participant (Session A found duplicated entries in multi-party wars).
	collate_targets = yes

	# Session A.2: an interaction_source_list candidate pool defeated collate_targets
	# (entries duplicated per enemy participant again), so the pool is back to
	# source = actor - the combination proven to give exactly one entry per location.
	# Cost: subject-owned locations are missing from the list for now, although the
	# vanilla Offer Tribute includes them (M4) - documented gap, revisit in v1.1.
	select_trigger = {
		looking_for_a = location
		source = actor
		target_flag = target
		visible = {
			bpt_giveable_location_trigger = yes
		}
	}

	# Can't fully annex the loser while also giving them land back.
	blocks_full_annexation = yes
	# Location targets exclusive with cede treaties: the same location can never
	# be demanded and returned in one deal.
	are_targets_exclusive = yes

	effect = {
		# Antagonism onto the RECEIVER, before the transfer so the location is
		# still valued from the giver's side. Same figure vanilla charges for
		# taking that location (bpt_given_location_antagonism).
		scope:target = {
			save_scope_as = bpt_given_location
		}
		scope:loser = {
			drop_antagonism_bomb = {
				target = scope:bpt_given_location
				modifier = antagonism_taking_land
				value = bpt_given_location_antagonism
			}
		}
		scope:target = {
			change_location_owner = scope:loser
		}
	}

	# v1: the AI never authors this treaty in its own offers.
	ai_desire = {
		value = 0
	}
}
