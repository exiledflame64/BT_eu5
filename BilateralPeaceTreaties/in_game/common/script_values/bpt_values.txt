# Bilateral Peace Treaties - shared values

# ============================================================================
# bpt_province_return_cost - what a whole province is worth to give back.
# ============================================================================
# Sum of the peace value of every location in scope:target that the winner
# actually owns, carrying the same taker-side wargoal factor as the
# single-location treaty. Iterator form copied from vanilla script values
# (byz_values.txt: every_owned_location = { limit = {...} add = ... }), and the
# bare data-function call is the documented scripted-value usage:
# "location_peace_cost(<giver>|<taker>)" (triggers.log).
bpt_province_return_cost = {
	value = 0
	scope:target = {
		every_location_in_province = {
			limit = {
				owner = scope:winner
			}
			add = {
				value = "location_peace_cost(scope:winner|scope:loser)"
			}
		}
	}
	multiply = bpt_taker_conquer_cost_factor
}

# ============================================================================
# ANTAGONISM MODEL - vanilla's receiver-antagonism for handing a location over.
# ============================================================================
# Vanilla itemises the figure as
#   Base 0.40  +  From Development  +  [GP Score of GIVER]  +  [GP Score of RECEIVER]
# where each great-power line is a percentage of Core (= Base + Development):
#   GIVER is a great power     -> -50% of Core   (FLAT: identical for England
#                                                  score 45.75 and France 111, so it
#                                                  keys off is_great_power, not score)
#   RECEIVER is a great power  -> +25% of Core
#   neither a great power      ->  no line
# Every combination measured (Sessions A.18-A.28): Eng(GP)->Scot(non)=Core x0.50;
# Fra(GP)->Eng(GP)=Core x0.75; Scot(non)->Eng(GP)=Core x1.25; non->non=Core. So
#   antagonism = Core x (1 - 0.5*giverGP + 0.25*receiverGP).
#
# SCOPE PROBLEM + SOLUTION. The EFFECT can read both parties (owner = giver,
# scope:loser = receiver), so bpt_given_location_antagonism computes the whole thing
# at once. The row LABEL cannot: a MakeScope.ScriptValue call is ROOT-ONLY (A.26),
# so a value called on the location sees `owner` (the giver) but never the receiver.
# The label therefore ADDS two parts with separate calls + loc arithmetic:
#   Core x (1 - 0.5*giverGP)   <- bpt_location_give_antagonism, called on the location
#   + Core x 0.25*receiverGP   <- bpt_location_core (on the location) TIMES
#                                 bpt_receiver_gp_factor (on TARGET_COUNTRY = the
#                                 losing side = the receiver; vanilla's own peace loc
#                                 confirms TARGET_COUNTRY gives to COUNTRY).
# Both routes reach the identical number, so shown == applied. Hence the split into
# bpt_location_core / bpt_location_give_antagonism / bpt_receiver_gp_factor.
#
# CORE = (8.09*dev + 53.15) / (dev + 69.33). Re-fitted (A.27) over nine measured
# pairs spanning development 6-36 (6->1.35 ... 36->3.27), within 0.03 everywhere. No
# sqrt exists in the modding surface (Add/Multiply/Divide only), so a ratio
# approximates vanilla's ~0.51*sqrt(dev) shape. Standing rule, earned four times:
# never fit until the samples span the range.

# Core alone (Base + Development), no great-power factor. Root = the location.
bpt_location_core = {
	value = development
	multiply = 8.09
	add = 53.15
	divide = {
		value = development
		add = 69.33
	}
}

# +0.25 weight iff the country this is called on is a great power, else 0. Root =
# a country: the RECEIVER (scope:loser in the effect, TARGET_COUNTRY in the label).
bpt_receiver_gp_factor = {
	value = 0
	if = {
		limit = { is_great_power = yes }
		add = 0.25
	}
}

# Core with the GIVER's -50% only (label part 1, and the province sum). Root = the
# location; `owner` is the giver - still winner-owned when read (the effect bombs
# before the transfer; the label reads the current owner). Confirmed to resolve in
# the label (A.27: a non-GP giver showed full Core, a GP giver showed half).
bpt_location_give_antagonism = {
	value = bpt_location_core
	if = {
		limit = { owner = { is_great_power = yes } }
		multiply = 0.5
	}
}

# ============================================================================
# bpt_given_location_antagonism - the FULL figure, for the EFFECT.
# ============================================================================
# The bomb's value is evaluated in scope:loser (the receiver). This hops into the
# saved location for Core + the giver, and reads scope:loser for the receiver - both
# reachable in effect scope, so the whole formula is one expression here.
bpt_given_location_antagonism = {
	value = 0
	scope:bpt_given_location = {
		add = {
			value = bpt_location_core
			multiply = {
				value = 1
				if = { limit = { owner = { is_great_power = yes } } subtract = 0.5 }
				if = { limit = { scope:loser = { is_great_power = yes } } add = 0.25 }
			}
		}
	}
}

# ============================================================================
# bpt_province_return_antagonism / bpt_province_core_sum - the province label's
# two summed parts.
# ============================================================================
# Both iterate every location in the province (root = the province, reached via
# MakeScope from the label). scope:winner is unresolved there, so no owner filter is
# possible (A.26); summing all locations is exact whenever the giver owns the whole
# province - the normal give-back case - and each location still carries its own
# owner's giver-factor, so a third-party split still reads sensibly.
# Part 1 = sum of the giver-term figure; Part 2 = sum of bare Core, which the label
# multiplies by bpt_receiver_gp_factor (called on TARGET_COUNTRY) to add the +25%.
bpt_province_return_antagonism = {
	value = 0
	every_location_in_province = {
		add = bpt_location_give_antagonism
	}
}

bpt_province_core_sum = {
	value = 0
	every_location_in_province = {
		add = bpt_location_core
	}
}

# ============================================================================
# bpt_indemnity_loan_value - gold worth one Loan to the paying side.
# ============================================================================
# Vanilla denominates peace gold in Loans: "Maximum is 5 Loans, totalling
# +717.27 (25 war score)" -> 1 Loan = 5% war score.
# A Loan is literally the country's borrowable loan, the figure on the Economy
# panel's Borrow button - measured in Session A.16: a country whose Borrow read
# 3.42 was allowed to offer at most 17.14 gold in the vanilla peace UI, i.e.
# 5 x 3.42. The earlier "five months of trade+tax income" reading matched one
# observation by coincidence and was wrong (it gave 13.31 per Loan here).
bpt_indemnity_loan_value = {
	value = scope:winner.country_estate_loan_size
}

# ============================================================================
# bpt_taker_conquer_cost_factor
# ============================================================================
# The engine multiplies a location's peace value by the TAKING side's
# conquer_cost factor of the war's wargoal (Session A.2: vanilla demand AND
# Offer Tribute both showed base x 1.5 under "Show Superiority").
#
# CONSTRAINT (Session A.4, error.log): the war_goal_type trigger accepts only
# the base war goal TYPES (readme list + destroy_army) - specific wargoal ids
# fail with "Trigger is simple assign, but assignment failed". So this table is
# PER TYPE, using the conquer_cost of each type's generic wargoal from
# common/wargoals/00_default.txt @1.3.11:
#
#   type                     source wargoal        attacker  defender
#   superiority              superiority           1.5       1 (omitted)
#   take_province            conquer_province      0.75      1
#   take_capital             take_capital          0.5       1
#   take_country             take_country          0.5       1
#   take_border              take_border           0.8       1
#   independence             independence          0.75      1
#   naval_superiority        naval                 10        1
#   enforce_military_access  demand_military_access 20       1
#   destroy_army             sengoku_wargoal       0.75      0.75
#   defend_capital           defend_capital        1         1 (no entry)
#
# EXACT for wars using those generic wargoals (incl. no-CB wars -> superiority).
# APPROXIMATE for exotic same-type variants (e.g. superiority_coalition 0.75/2.5,
# superiority_livonian_wars 0.75/0.75, fabricated_conquer_province 1.3,
# take_capital_claim_throne 2/0.25...). If exact parity for those ever matters:
# the war's casus_belli IS script-checkable (casus_belli ?= casus_belli:<key>)
# and CBs pin their wargoal via war_goal_type = <id>, so a CB->factor override
# table can be layered on top. Deferred until the author asks.
#
# Calling context: peace treaty cost (scope:war/winner/loser set). Taker of a
# returned location = scope:loser; its side picks attacker/defender values
# ("is_an_attacker": war-scope trigger, country target).

bpt_taker_conquer_cost_factor = {
	value = 1

	# --- attacker-only factors (defender omitted -> 1) ---
	if = {
		limit = { scope:war = { war_goal_type = superiority is_an_attacker = scope:loser } }
		multiply = 1.5
	}
	if = {
		limit = { scope:war = { war_goal_type = take_province is_an_attacker = scope:loser } }
		multiply = 0.75
	}
	if = {
		limit = { scope:war = { war_goal_type = take_capital is_an_attacker = scope:loser } }
		multiply = 0.5
	}
	if = {
		limit = { scope:war = { war_goal_type = take_country is_an_attacker = scope:loser } }
		multiply = 0.5
	}
	if = {
		limit = { scope:war = { war_goal_type = take_border is_an_attacker = scope:loser } }
		multiply = 0.8
	}
	if = {
		limit = { scope:war = { war_goal_type = independence is_an_attacker = scope:loser } }
		multiply = 0.75
	}
	if = {
		limit = { scope:war = { war_goal_type = naval_superiority is_an_attacker = scope:loser } }
		multiply = 10
	}
	if = {
		limit = { scope:war = { war_goal_type = enforce_military_access is_an_attacker = scope:loser } }
		multiply = 20
	}

	# --- symmetric ---
	if = {
		limit = { scope:war = { war_goal_type = destroy_army } }
		multiply = 0.75
	}
}
