# Bilateral Peace Treaties - shared values

# =======================================================================================================================
# bpt_province_return_cost - what a whole province is worth to give back.
# =======================================================================================================================
# Sum of the peace value of every location in scope:target that the winner actually owns,
# carrying the same taker-side wargoal factor as the single-location treaty.
# Iterator form copied from vanilla script values (byz_values.txt: every_owned_location = { limit = {...} add = ... }),
# and the bare data-function call is the documented scripted-value usage:
# "location_peace_cost(<giver>|<taker>)" (triggers.log).

bpt_province_return_cost = {
	value = 0
	scope:target = {
		every_location_in_province = {
			limit = {
				owner = scope:winner
			}
			add = {
				value = "location_peace_cost(scope:winner|scope:loser)"
			}
		}
	}
	multiply = bpt_taker_conquer_cost_factor
}

# =======================================================================================================================
# ANTAGONISM MODEL - vanilla's receiver-antagonism for handing a location over.
# =======================================================================================================================
# Vanilla itemises the figure as
#   Base 0.40  +  Development  +  [GP Score of GIVER]  +  [GP Score of RECEIVER]
# Where each great-power line is a percentage of Core (= Base + Development):
#   GIVER is a great power     -> -50% of Core (flat)
#   RECEIVER is a great power  -> +25% of Core
#   neither a great power      ->  no line
#   
# So antagonism = Core x (1 - 0.5*giverGP + 0.25*receiverGP).
#
# SCOPE PROBLEM + SOLUTION. The EFFECT can read both parties (owner = giver, scope:loser = receiver),
# so bpt_given_location_antagonism computes the whole thing at once.
# The row LABEL cannot: a MakeScope.ScriptValue call is ROOT-ONLY,
# so a value called on the location sees `owner` (the giver) but never the receiver.
# The label therefore ADDS two parts with separate calls + loc arithmetic:
#   Core x (1 - 0.5*giverGP)   <- bpt_location_give_antagonism, called on the location
#   + Core x 0.25*receiverGP   <- bpt_location_core (on the location) TIMES
#                                 bpt_receiver_gp_factor (on TARGET_COUNTRY = the
#                                 losing side = the receiver; vanilla's own peace loc
#                                 confirms TARGET_COUNTRY gives to COUNTRY).
# Both routes reach the identical number, so shown == applied. Hence the split into
# bpt_location_core / bpt_location_give_antagonism / bpt_receiver_gp_factor.
#
# CORE = (8.09 × dev(loc) + 53.15) / (dev(loc) + 69.33). Measured by 9 points on the graph for dev values 6-36:
# (6->1.35 ... 36->3.27), within 0.03 everywhere. This ratio approximates vanilla's ~0.51*sqrt(dev) shape.

bpt_location_core = {
	value = development
	multiply = 8.09
	add = 53.15
	divide = {
		value = development
		add = 69.33
	}
}

# +0.25 weight if the country is a great power, else 0. Root = a country: the RECEIVER 
# (scope:loser in the effect, TARGET_COUNTRY in the label).

bpt_receiver_gp_factor = {
	value = 0
	if = {
		limit = { is_great_power = yes }
		add = 0.25
	}
}

# Core with the GIVER's -50% only. Root = the location; `owner` is the giver - still winner-owned when read.

bpt_location_give_antagonism = {
	value = bpt_location_core
	if = {
		limit = { owner = { is_great_power = yes } }
		multiply = 0.5
	}
}

# =======================================================================================================================
# bpt_given_location_antagonism - the FULL figure, for the EFFECT.
# =======================================================================================================================
# The bomb's value is evaluated in scope:loser (the receiver). This hops into the saved location for Core + the giver,
# and reads scope:loser for the receiver - both reachable in effect scope, so the whole formula is one expression here.

bpt_given_location_antagonism = {
	value = 0
	scope:bpt_given_location = {
		add = {
			value = bpt_location_core
			multiply = {
				value = 1
				if = { limit = { owner = { is_great_power = yes } } subtract = 0.5 }
				if = { limit = { scope:loser = { is_great_power = yes } } add = 0.25 }
			}
		}
	}
}

# =======================================================================================================================
# bpt_province_return_antagonism / bpt_province_core_sum - the province label's two summed parts.
# =======================================================================================================================
# Both iterate every location in the province (root = the province, reached via MakeScope from the label). 
# scope:winner is unresolved there, so no owner filter is possible; 
# summing all locations is exact whenever the giver owns the whole province, and each location still carries its own owner's giver-factor,
# so a third-party split still reads sensibly.
# Part 1 = sum of the giver-term figure; Part 2 = sum of bare Core, which the label
# multiplies by bpt_receiver_gp_factor (called on TARGET_COUNTRY) to add the +25%.

bpt_province_return_antagonism = {
	value = 0
	every_location_in_province = {
		add = bpt_location_give_antagonism
	}
}

bpt_province_core_sum = {
	value = 0
	every_location_in_province = {
		add = bpt_location_core
	}
}

# =======================================================================================================================
# bpt_indemnity_loan_value - gold worth one Loan to the paying side.
# =======================================================================================================================
# Vanilla denominates peace gold in Loans: "Maximum is 5 Loans, totalling (25 war score)" -> 1 Loan = 5% war score.
# 'A Loan' is the country's borrowable loan, the figure on the Economy panel's Borrow button.

bpt_indemnity_loan_value = {
	value = scope:winner.country_estate_loan_size
}

# =======================================================================================================================
# bpt_taker_conquer_cost_factor
# =======================================================================================================================
# The engine multiplies a location's peace value by the TAKING side's conquer_cost factor of the war's wargoal 
#
# CONSTRAINT: the war_goal_type trigger accepts only the base war goal TYPES (readme list + destroy_army)
# specific wargoal ids fail with "Trigger is simple assign, but assignment failed". So this table is PER TYPE,
# using the conquer_cost of each type's generic wargoal from common/wargoals/00_default.txt @1.3.11:
#
#   type                     source wargoal        attacker  defender
#   superiority              superiority           1.5       1 (omitted)
#   take_province            conquer_province      0.75      1
#   take_capital             take_capital          0.5       1
#   take_country             take_country          0.5       1
#   take_border              take_border           0.8       1
#   independence             independence          0.75      1
#   naval_superiority        naval                 10        1
#   enforce_military_access  demand_military_access 20       1
#   destroy_army             sengoku_wargoal       0.75      0.75
#   defend_capital           defend_capital        1         1 (no entry)
#
# EXACT for wars using those generic wargoals (incl. no-CB wars -> superiority).
# APPROXIMATE for exotic same-type variants (e.g. superiority_coalition 0.75/2.5,
# superiority_livonian_wars 0.75/0.75, fabricated_conquer_province 1.3, take_capital_claim_throne 2/0.25...).
# If exact parity for those ever matters: the war's casus_belli IS script-checkable (casus_belli ?= casus_belli:<key>)
# and CBs pin their wargoal via war_goal_type = <id>, so a CB->factor override table can be layered on top.
#
# Calling context: peace treaty cost (scope:war/winner/loser set). Taker of a returned location = scope:loser;
# its side picks attacker/defender values ("is_an_attacker": war-scope trigger, country target).

bpt_taker_conquer_cost_factor = {

	# The engine multiplies a location's peace value by the TAKING side's conquer_cost factor of the war's wargoal. 
	# conquer_cost is NOT script-readable and war_goal_type only exposes the BASE type, so this has two layers:
	#   1. 	CB OVERRIDES below - matched on the war's own casus belli (scope:war.casus_belli),
	#		for the ~36 single-wargoal vanilla CBs whose conquer_cost differs from its base type. 
	#	   	Exact for those. Generated from common/wargoals @1.3.11.
	#   2. 	The generic base-type table in the final `else` - for else.
	# 
	# taker = scope:loser; is_an_attacker picks the attacker vs defender cost.
	# NOT covered (fall through to the base table, an approximation): multi-wargoal CBs
	# (the war doesn't expose which wargoal it picked - e.g. cb_subjugation, cb_tribal_feud,
	# cb_league_of_public_weal, attack_threat, cb_red_turban_rebellions_against_other_rebels);
	# cb_restore_roman_borders (scripted conquer_cost); and wargoals that OMIT conquer_cost.

	if = {
		limit = { scope:war = { OR = {
			casus_belli ?= casus_belli:cb_red_turban_rebellions
			casus_belli ?= casus_belli:cb_red_turban_rebellions_rein_in_rebels
		} } }
		value = 0.02
	}
	else_if = {
		limit = { scope:war = { OR = {
			casus_belli ?= casus_belli:cb_hegemon
			casus_belli ?= casus_belli:cb_imperialism
			casus_belli ?= casus_belli:spread_the_revolution_cb
		} } }
		value = 0.25
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_high_kingship_reclamation } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 0.25 }
		else = { add = 10 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_nationalist } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 0.2 }
		else = { add = 1 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_expansion_into_the_steppes } }
		value = 0.5
	}
	else_if = {
		limit = { scope:war = { OR = {
			casus_belli ?= casus_belli:cb_dlh_invasion_of_china
			casus_belli ?= casus_belli:cb_reclaim_andalucia
		} } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 0.5 }
		else = { add = 0.6 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_push_back_colonizers } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 0.5 }
		else = { add = 10.0 }
	}
	else_if = {
		limit = { scope:war = { OR = {
			casus_belli ?= casus_belli:cb_hundred_years_war
			casus_belli ?= casus_belli:cb_hundred_years_war_fra
			casus_belli ?= casus_belli:cb_livonian_wars
			casus_belli ?= casus_belli:cb_shed_shackles_of_the_ilkahante
			casus_belli ?= casus_belli:cb_unify_ilkahante
		} } }
		value = 0.75
	}
	else_if = {
		limit = { scope:war = { OR = {
			casus_belli ?= casus_belli:cb_excommunication
			casus_belli ?= casus_belli:cb_lat_reconquest
			casus_belli ?= casus_belli:cb_sco_invade_england
		} } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 0.75 }
		else = { add = 1 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_coalition } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 0.75 }
		else = { add = 2.5 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:great_northern_conflict } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 0.9 }
		else = { add = 1 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_dissolve_tatar_yoke } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 1.0 }
		else = { add = 0.75 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_parliament_conquer_province } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 1.25 }
		else = { add = 1 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_deus_vult } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 1.2 }
		else = { add = 1 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_gag_force_country_into_faction } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 1.5 }
		else = { add = 1.25 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_sound_toll_exemptions_broken } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 1.5 }
		else = { add = 1 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_heretic } }
		value = 1.75
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_flower_wars } }
		value = 10.0
	}
	else_if = {
		limit = { scope:war = { OR = {
			casus_belli ?= casus_belli:cb_humiliate
			casus_belli ?= casus_belli:cb_humiliate_rival
			casus_belli ?= casus_belli:cb_slave_raiding
		} } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 10.0 }
		else = { add = 1 }
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_high_kingship_subjugate } }
		value = 100.0
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_claim_throne } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 2 }
		else = { add = 0.25 }
	}
	else_if = {
		limit = { scope:war = { OR = {
			casus_belli ?= casus_belli:cb_liberate_hre_member
			casus_belli ?= casus_belli:cb_prevent_annexation
		} } }
		value = 2
	}
	else_if = {
		limit = { scope:war = { casus_belli ?= casus_belli:cb_high_kingship_overthrow } }
		value = 0
		if = { limit = { scope:war = { is_an_attacker = scope:loser } } add = 2 }
		else = { add = 4 }
	}
	else = {
		# --- generic base-type table (attacker-only factors; defender omitted -> 1) ---
		value = 1
		if = {
			limit = { scope:war = { war_goal_type = superiority is_an_attacker = scope:loser } }
			multiply = 1.5
		}
		if = {
			limit = { scope:war = { war_goal_type = take_province is_an_attacker = scope:loser } }
			multiply = 0.75
		}
		if = {
			limit = { scope:war = { war_goal_type = take_capital is_an_attacker = scope:loser } }
			multiply = 0.5
		}
		if = {
			limit = { scope:war = { war_goal_type = take_country is_an_attacker = scope:loser } }
			multiply = 0.5
		}
		if = {
			limit = { scope:war = { war_goal_type = take_border is_an_attacker = scope:loser } }
			multiply = 0.8
		}
		if = {
			limit = { scope:war = { war_goal_type = independence is_an_attacker = scope:loser } }
			multiply = 0.75
		}
		if = {
			limit = { scope:war = { war_goal_type = naval_superiority is_an_attacker = scope:loser } }
			multiply = 10
		}
		if = {
			limit = { scope:war = { war_goal_type = enforce_military_access is_an_attacker = scope:loser } }
			multiply = 20
		}
		# --- symmetric ---
		if = {
			limit = { scope:war = { war_goal_type = destroy_army } }
			multiply = 0.75
		}
	}
}
