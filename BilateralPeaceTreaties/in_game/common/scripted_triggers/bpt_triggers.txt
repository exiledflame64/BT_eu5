# Bilateral Peace Treaties - shared triggers
# Peace treaty scopes: scope:winner = taker, scope:loser = giver, scope:war = the war
# (documented in game/in_game/common/peace_treaties/readme.txt)

# Ensure it's only Player VS AI, used in every bpt_ treaty's potential -> the treaties never exist in AI-AI peace deals.

bpt_human_involved_trigger = {
	OR = {
		scope:winner = { is_ai = no }
		scope:loser = { is_ai = no }
	}
}

# Which locations are offerable back to the loser.
# ROOT = the candidate location; inside select_trigger: scope:actor = winner (taker), scope:recipient = loser (giver).
# Mirrors the vanilla Offer Tribute: ALL locations of the giving side are offerable - own land incl. capital,
# plus subjects' land ("is_subject_of": triggers.log docs).

bpt_giveable_location_trigger = {
	OR = {
		owner = scope:actor
		owner = {
			is_subject_of = scope:actor
		}
	}
}
